```tsx
// Web: React, consuming @dizyx/nockerl-tokens. (Web is the laggard platform. Spec + live demo,
// not yet exported from @dizyx/nockerl-react; promotion is tracked.)
// Renders a whole markdown string as styled prose: headings bind to the type
// ramp, inline + fenced code reuse <CodeBlock>, links reuse <NockerlLink>.

export function AssistantMessage({ markdown }: { markdown: string }) {
  return (
    <MarkdownContent
      content={markdown}
      density="compact"          // chat-bubble density; "comfortable" for docs
      onImageClick={openLightbox}
    />
  );
}
```

```kotlin
// Android (Jetpack Compose, canonical). chat/ui/MarkdownContent.kt wraps the
// mikepenz multiplatform-markdown-renderer with the Nockerl type ramp + colors.
// StreamingMarkdownContent renders LIVE markdown flash-free during SSE streaming.
import com.nockerl.app.chat.ui.StreamingMarkdownContent

StreamingMarkdownContent(
    text = block.text,
    isStreaming = isStreaming,
    onImageClick = { url -> onImageClick(url) },
    textColor = if (isUser) colors.onUserCard else colors.onCardAlt,
)

// The shared styling, applied by MarkdownBody to both the streaming + final paths:
Markdown(
    rememberMarkdownState(content = text, immediate = true),   // synchronous parse, no empty frame
    colors = markdownColor(
        text = resolvedTextColor,
        codeBackground = resolvedTextColor.copy(alpha = 0.07f),        // recessed code well
        inlineCodeBackground = resolvedTextColor.copy(alpha = 0.10f),  // inline code chip
    ),
    typography = markdownTypography(
        h1 = MaterialTheme.typography.titleMedium,   // compact (chat) heading ramp
        h2 = MaterialTheme.typography.titleSmall,
        h3 = MaterialTheme.typography.labelLarge,
        h4 = MaterialTheme.typography.labelLarge,
        text = MaterialTheme.typography.bodyMedium.copy(lineHeight = 22.sp),
        code = MaterialTheme.typography.bodySmall.copy(fontFamily = FontFamily.Monospace),
    ),
    padding = markdownPadding(block = 6.dp, list = 8.dp, listItemBottom = 6.dp, listIndent = 8.dp),
    components = markdownComponents(codeBlock = highlightedCodeBlock, codeFence = highlightedCodeFence),
    animations = markdownAnimations { this },       // disable the lib's per-text size spring
)
```

<Aside type="caution" title="Cross-platform drift (tracked for reconciliation)">
- **Swift has no markdown surface.** Nockerl Voice is a dictation app; transcripts
  are plain `Text(record.text)`. There is no styled-prose renderer to mirror. The
  Swift code/params above are the *intended* API, flagged here.
- **Heading scale** *(resolved in [adjudication 2026-07-02 · B10])*: markdown ships a **shared
  two-density ramp**: compact (chat) + comfortable (docs). Android (today the compact ramp only:
  h1 → `title.medium`, h2 → `title.small`, h3/h4 → `label.large`) gains the comfortable ramp; the
  demo's docs density is that shared second density, not a web-only extension.
- **Code well** *(resolved in [adjudication 2026-07-02 · B10])*: code wells use the **`canvas-alt`
  recessed token, one definition shared with [Code block](/components/code-block/)**. Android's
  alpha-washed inline (10%) / fenced (7%) backgrounds give way to the shared recessed-well token.
</Aside>
