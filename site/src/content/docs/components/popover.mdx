// Web: React, consuming @dizyx/nockerl-tokens. (Web is the laggard platform;
// this is the canonical API the published @dizyx/nockerl-react package exposes.)
// A popover is a NON-MODAL anchored panel of free-form content: not a menu,
// not a tooltip, not a dialog.
import { NockerlPopover } from '@dizyx/nockerl-react';

export function SessionHeader() {
  return (
    <NockerlPopover
      placement="bottom"
      arrow
      trigger={<button aria-label="Profile">PM</button>}
    >
      {({ close }) => (
        <>
          <h3 className="title">the design lead</h3>
          <p className="muted">Owner · dizyx</p>
          <button onClick={close}>View full profile</button>
        </>
      )}
    </NockerlPopover>
  );
}

```kotlin
// Android: Jetpack Compose. NO NockerlPopover ships yet; this is the INTENDED
// API, built on Material 3 Popup + the reserved tier-3 popover surface
// (NockerlSurface, NockerlElevation.Level3, NockerlPanelShape). See the drift note.
@Composable
fun NockerlPopover(
    expanded: Boolean,
    onDismissRequest: -> Unit,
    anchorEdge: PopoverEdge = PopoverEdge.Bottom,
    arrow: Boolean = true,
    content: @Composable ColumnScope.() -> Unit,
) {
    if (!expanded) return
    Popup(onDismissRequest = onDismissRequest, properties = PopupProperties(focusable = true)) {
        NockerlSurface(
            shape = NockerlPanelShape,
            gradient = nockerlSurfaceGradient(),     // cardSurface2 → cardSurface1
            elevation = NockerlElevation.Level3,      // reserved for popovers
            border = LocalNockerlColors.current.cardHairline,
        ) {
            Column(Modifier.padding(16.dp), content = content)
        }
    }
}
```

<Aside type="note" title="Popover vs. menu vs. tooltip vs. dialog">
Four anchored/floating siblings. Pick by content and modality:
- **Popover** (this page) is a **non-modal anchored panel of free-form content**: a profile card, a small form, a filter panel. Opens on click/focus, points a beak at its trigger, never dims the ground.
- **[Menu](/components/menu/)** is the same anchor + beak machinery but holds a **list of action items** (icons, shortcuts, checks), not arbitrary content.
- **[Tooltip](/components/tooltip/)** is a **tiny, non-interactive hover hint** (one line, no controls, no focus). When you need a field or a button inside, it's a popover.
- **[Dialog](/components/dialog/)** is **modal and centered**: it dims the ground and traps focus until resolved, and is *not* anchored to a trigger.
</Aside>

<Aside type="caution" title="Cross-platform drift (tracked for reconciliation)">
- **Android has no shipped popover.** `NockerlSurface.kt` reserves tier-3 elevation for "popovers", but no `NockerlPopover` exists yet; the Kotlin sample is the intended API on Material `Popup` (`RichTooltip` / `TooltipBox` are the nearest stock substrate). Drift until built.
- **Beak:** web + macOS draw a beak; the intended Android `Popup` would need a custom one (Material `Popup` has none).
- **Width:** macOS `InfoTip` fixes `width: 250`; web sizes per content. Tokenize a shared popover width when Android lands.
</Aside>
