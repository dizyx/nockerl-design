// Android (Jetpack Compose, canonical). The shipped single-select dropdown is a
// DropdownAnchor (recessed control-shaped field + ArrowDropDown) over a Material3
// DropdownMenu of DropdownMenuItem rows. See chat/ui/SessionCreationDropdowns.kt.
import androidx.compose.material3.DropdownMenu
import androidx.compose.material3.DropdownMenuItem
import androidx.compose.material3.Text
import com.nockerl.app.chat.ui.DropdownAnchor

@Composable
fun ProviderDropdown(
    providers: List<ProviderInfo>,
    selectedProvider: ProviderInfo?,
    isLoading: Boolean,
    onSelect: (ProviderInfo) -> Unit,
) {
    var expanded by remember { mutableStateOf(false) }

    Box {
        DropdownAnchor(
            label = selectedProvider?.name ?: "Select a provider",
            placeholder = selectedProvider == null,
            isLoading = isLoading,
            onClick = { if (!isLoading && providers.isNotEmpty()) expanded = true },
        )
        DropdownMenu(expanded = expanded, onDismissRequest = { expanded = false }) {
            providers.forEach { provider ->
                DropdownMenuItem(
                    text = { Text(provider.name) },
                    onClick = { onSelect(provider); expanded = false },
                )
            }
        }
    }
}

// macOS: SwiftUI (canonical). The shipped single-select is a Menu of Buttons; the
// chosen row carries a leading checkmark via Label(systemImage:). The trigger is a
// chevron.up.chevron.down on a recessed accent-outlined field. See UI/AppSettingsView.swift.
Menu {

        micMenuLabel("System Default", selected: settings.selectedMicUID.isEmpty)
    }
    ForEach(micMonitor.devices) { mic in
        Button { settings.selectedMicUID = mic.uid } label: {
            micMenuLabel(mic.name, selected: settings.selectedMicUID == mic.uid)
        }
    }
} label: {
    HStack(spacing: 8) {
        Text(currentMicName).lineLimit(1).truncationMode(.middle)
        Spacer(minLength: 0)
        Image(systemName: "chevron.up.chevron.down")
    }
    .foregroundStyle(NockerlTheme.accent)
    .padding(.horizontal, 12).padding(.vertical, 8)
    .background(NockerlTheme.canvasAlt, in: RoundedRectangle(cornerRadius: 12))
    .overlay(RoundedRectangle(cornerRadius: 12).strokeBorder(NockerlTheme.accent, lineWidth: 1.5))
}
.menuStyle(.borderlessButton)
.menuIndicator(.hidden)

// A checkmark on the selected row; a plain Text otherwise.
@ViewBuilder
func micMenuLabel(_ name: String, selected: Bool) -> some View {
    if selected { Label(name, systemImage: "checkmark") } else { Text(name) }
}

<Aside type="note" title="Pick the right picker">
Select is the **plainest** picker: one value from a fixed list, no typing. Reach for a
neighbour when you need more: [Combobox](/components/combobox/) when the user should
**type to filter** a long list (autocomplete); [Multi-select](/components/multi-select/)
when they pick **several** values (token chips, per-row checkboxes, select-all);
[Menu](/components/menu/) when the rows are **actions**, not a value (icons, shortcuts,
submenus).
</Aside>

<Aside type="caution" title="Cross-platform drift (tracked for reconciliation)">
The spec is in review; these are the real divergences between the shipped platforms:

- **Selected-row check** *(resolved in [adjudication 2026-07-02 · C5])*: a **trailing check** on
  the selected row is the shared expression. Swift + web already stamp it; Compose adds a trailing
  check to its `DropdownMenuItem` (today the selected value reads only in the anchor).
- **Control radius:** Compose uses `NockerlControlShape` (**12dp**, the token); Swift's
  mic picker hardcodes an **8pt** `RoundedRectangle`; it should consume `radius.control`.
- **Grouping + secondary text:** web supports `group` headers and a `secondary` line per
  option; the shipped Android/Swift pickers list flat label-only rows.
- **Sizing:** web has `size` (sm/md); Compose's anchor is fixed at 48dp; Swift is intrinsic.
- **Error state:** web has a first-class `errorText` (warm border + icon + message); the
  apps surface validation outside the control today.
</Aside>
