```tsx
// Web: React, consuming @dizyx/nockerl-tokens. (Web is the laggard platform;
// this is the intended API, not yet exported from @dizyx/nockerl-react (promotion tracked).
// Neither app ships a custom calendar, so the web demo is designed from the laws.)

export function ScheduleForm() {
  const [due, setDue] = useState<Date | null>(null);
  const [span, setSpan] = useState<{ start: Date | null; end: Date | null }>({ start: null, end: null });

  return (
    <>
      {/* Field + popover, single date, no past dates. */}
      <DatePicker label="Due date" value={due} onChange={setDue} min={new Date()} />

      {/* Always-open inline calendar. */}
      <DatePicker label="Run on" value={due} onChange={setDue} inline />

      {/* Range mode: two-endpoint range with in-range tint (absorbs the former DateRangePicker). */}
      <DatePicker range label="Window" value={span} onChange={setSpan} />
    </>
  );
}
```

```kotlin
// Android: Jetpack Compose, Material 3 (BOM 2026.02.01). The app has no custom
// calendar: tasks/domain/Task.kt stores `dueDate` as an ISO String and only
// FORMATS it (DateTimeFormatter "MMM d, yyyy h:mm a"). Use the stock picker,
// themed by NockerlTheme (cyan accent, 12dp control radius).
import androidx.compose.material3.DatePicker
import androidx.compose.material3.DatePickerDialog
import androidx.compose.material3.rememberDatePickerState

val state = rememberDatePickerState(
    initialSelectedDateMillis = task.dueDateMillis,
    selectableDates = NoPastDates,   // SelectableDates: gate min/max + per-day
)

DatePickerDialog(
    onDismissRequest = onDismiss,
    confirmButton = {
        NockerlButton("Set", onClick = { onPick(state.selectedDateMillis) })
    },
) {
    DatePicker(state = state, title = null)
}
```

<Fragment slot="swift">

```swift
// macOS: SwiftUI. Nockerl Voice has no custom calendar; it formats dates with
// `.dateTime` / `Calendar.current` (HistoryView, HomeSection). Use the stock
// graphical DatePicker, tinted to the brand cyan from NockerlTheme.
import SwiftUI

struct ScheduleForm: View {
    @State private var due = Date()

    var body: some View {
        DatePicker(
            "Due date",
            selection: $due,
            in: Date()...,                    // min bound; use a ClosedRange for min…max
            displayedComponents: .date
        )
        .datePickerStyle(.graphical)          // the month-grid calendar
        .tint(NockerlTheme.accent)            // cyan selection fill
    }
}
```

**Range mode** (`range`) swaps `value`/`onChange` for `value: { start: Date | null; end: Date | null }`
and a matching `onChange`; the two endpoints get the accent fill and the days between get the
in-range tint. (The former standalone `DateRangePicker` was merged into this range mode;
recover its preset-rail / dual-month demo from git history if that richer surface is revived.)

<DocCallout tone="warning" title="Cross-platform drift (tracked for reconciliation)">
- **No shipped custom calendar.** Neither Android nor Voice draws its own month grid.
  Android stores `dueDate` as an ISO string and only *formats* it; Voice formats a chart
  axis. Kotlin/Swift above document the **stock** Material 3 / SwiftUI pickers; the web
  island is designed originally from the laws (recessed field-trigger, lifted calendar card,
  cyan selected fill, today ring, in-range tint).
- **Range:** web ships range as a **mode of `DatePicker`** (the standalone `DateRangePicker` was
  merged in, ); SwiftUI has no range picker (model it as two `DatePicker`s); Material 3 has
  `DateRangePicker` but the apps don't use it yet.
- **Week start** *(resolved)*: `weekStartsOn` **defaults to the
  device locale**, with an explicit prop override. The Nockerl web demo (Monday-first) and the
  stock pickers converge on that shared default + prop.
</DocCallout>
