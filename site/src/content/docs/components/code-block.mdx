// Web: React, consuming @dizyx/nockerl-tokens. (Web is the laggard platform. Spec + live demo,

export function FileViewer({ source }: { source: string }) {
  return (

      filename="backoff.ts"
      language="typescript"
      code={source}
      showLineNumbers
      highlightLines={[4]}
      onCopy={() => track('code.copy')}
    />
  );

// Header chrome (filename + language pill) is supplied by the surrounding sheet:

// The recessed, 2D-scrollable monospace well (line-number gutter; lines never wrap):
FileViewerContent(
    content = state.viewerContent,   // FileContent(content, language, isBinary, error)
    isLoading = state.isViewerLoading,
    modifier = Modifier.fillMaxSize(),

// Unified diff for an Edit/Write (red/green lines, +/- gutter, copy of the new content):
val diffLines = computeDiff(old = before, new = after)   // List<DiffLine>

```swift
// macOS: SwiftUI, from the NockerlDesign SwiftPM package. `NockerlCodeBlock` is a
// RECESSED monospace well (code sinks into the page, law §2): an optional filename
// header that also hosts the copy button, an optional line-number gutter, and
// horizontal scrolling. Lines NEVER wrap unless `wrap` is set. Replaces Voice's raw
// `.monospaced` Text. The `code` argument is the first positional.
import NockerlDesign

NockerlCodeBlock(
    source,                       // the source text, rendered verbatim
    filename: "backoff.swift",    // header title + copy button; omit → no header
    showLineNumbers: true
)

// No filename → the well renders without a header (copy still appears where the
// platform has a pasteboard).
NockerlCodeBlock(snippet)

// Soft-wrap long lines instead of scrolling horizontally.
NockerlCodeBlock(longLine, wrap: true)

// Syntax highlighting, highlight-lines, and the unified-diff mode are documented
// web/Android-first follow-ups, not in NockerlDesign yet.
```

<Aside type="caution" title="Cross-platform drift (tracked for reconciliation)">
These are the real divergences:

- **Swift now ships the well.** `NockerlCodeBlock` (NockerlDesign) provides the recessed
  well, filename header, copy, line-number gutter, and horizontal scroll, replacing
  Voice's raw `.monospaced` `Text`. Syntax highlighting, highlight-lines, and diff are
  **not** in it yet (web/Android-first follow-ups).
- **Mono font token:** the ratified code face is **Space Mono**, but the Swift token build
  emits no mono type role and the package bundles no fonts, so `NockerlCodeBlock` renders
  the **platform** `.monospaced` design. Emitting + bundling the mono token is the flagged
  follow-up.
- **Syntax highlighting:** the Android file viewer is **plain monospace** (highlighting
  deferred), and `NockerlCodeBlock` is too; only the markdown renderer + this web demo tint
  syntax. The well + chrome are identical either way. Pick one policy.
- **Language pill:** `NockerlCodeBlock` takes **no** `language` (no pill), and there is also no
  `file-type.swift` token. Add both when the language affordance lands on Swift.
- **Diff** ships on **Android** (`DiffView`) and is specified for web; **Swift** has none.
</Aside>
